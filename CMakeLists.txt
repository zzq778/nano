cmake_minimum_required(VERSION 3.10)
project(rescue_vision_standalone LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc imgcodecs)
set(ASTRA_SDK_ROOT "" CACHE PATH "Explicit matching Orbbec SDK installation root")
if(ASTRA_SDK_ROOT)
  set(ASTRA_INCLUDE "${ASTRA_SDK_ROOT}/include/astra_camera/openni2")
  set(ASTRA_LIBRARY "${ASTRA_SDK_ROOT}/lib/libOpenNI2.so")
  if(NOT EXISTS "${ASTRA_INCLUDE}/OpenNI.h" OR NOT EXISTS "${ASTRA_LIBRARY}")
    message(FATAL_ERROR "Astra requires the matching headers and library under ASTRA_SDK_ROOT")
  endif()
  add_library(vision_astra SHARED src/astra_camera.cpp)
  target_include_directories(vision_astra PRIVATE "${ASTRA_INCLUDE}")
  target_link_libraries(vision_astra PRIVATE "${ASTRA_LIBRARY}")
  target_compile_definitions(vision_astra PRIVATE RESCUE_ASTRA_REDIST="${ASTRA_SDK_ROOT}/lib")
  set_target_properties(vision_astra PROPERTIES BUILD_RPATH "${ASTRA_SDK_ROOT}/lib")
  target_compile_options(vision_astra PRIVATE -O3 -Wall -Wextra)
endif()
add_library(vision_green SHARED src/v12_green.cpp src/green_batch_bridge.cpp src/green_session.cpp src/topface.cpp src/topface_bridge.cpp)
target_include_directories(vision_green PRIVATE src include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_green PRIVATE ${OpenCV_LIBS})
target_compile_options(vision_green PRIVATE -O3 -Wall -Wextra)
add_library(vision_orange SHARED src/orange_native.cpp src/orange_bridge.cpp)
target_include_directories(vision_orange PRIVATE src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_orange PRIVATE ${OpenCV_LIBS})
target_compile_options(vision_orange PRIVATE -O3 -Wall -Wextra)
add_library(vision_metric SHARED src/metric_bridge.cpp vendor_model/src/model_geometry.cpp)
target_include_directories(vision_metric PRIVATE vendor_model/include include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vision_metric PRIVATE ${OpenCV_LIBS})
target_compile_options(vision_metric PRIVATE -O3 -Wall -Wextra)
option(WITH_TENSORRT "Build the six-class TensorRT frontend" ON)
if(WITH_TENSORRT)
  find_path(TENSORRT_INCLUDE NvInfer.h PATHS /usr/include/aarch64-linux-gnu)
  find_library(TENSORRT_LIB nvinfer)
  find_library(CUDART_LIB cudart PATHS /usr/local/cuda/lib64)
  if(NOT TENSORRT_INCLUDE OR NOT TENSORRT_LIB OR NOT CUDART_LIB)
    message(FATAL_ERROR "TensorRT/CUDA missing; use -DWITH_TENSORRT=OFF for the color route only")
  endif()
  add_library(vision_detector SHARED src/detector_bridge.cpp src/tensorrt_engine.cpp src/yolo_raw_adapter.cpp src/detection_postprocessor.cpp)
  target_compile_definitions(vision_detector PRIVATE RESCUE_VISION_HAS_TENSORRT=1)
  target_include_directories(vision_detector PRIVATE include ${OpenCV_INCLUDE_DIRS} ${TENSORRT_INCLUDE} /usr/local/cuda/include)
  target_link_libraries(vision_detector PRIVATE ${OpenCV_LIBS} ${TENSORRT_LIB} ${CUDART_LIB})
  target_compile_options(vision_detector PRIVATE -O3 -Wall -Wextra)
endif()
